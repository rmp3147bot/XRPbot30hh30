def load_json(path, default):
    if not os.path.exists(path):
        return default
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception as exc:
        log.error("Cannot load %s: %s", path, exc)
        return default


def save_json(path, value):
    temp = path + ".tmp"
    with open(temp, "w", encoding="utf-8") as f:
        json.dump(value, f, indent=2, ensure_ascii=False)
    os.replace(temp, path)


def append_jsonl(path, value):

# XT PARAMETER CANONICALIZATION
# ============================================================================

def canonical_form(params):
    if not params:
        return ""
    pairs = []
    for key in sorted(params.keys()):
        value = params[key]
        if value is None:
            continue
        if isinstance(value, bool):
            value = "true" if value else "false"
        pairs.append(f"{key}={value}")
    return "&".join(pairs)


# ============================================================================
# XT SIGNATURE
# ============================================================================

def calculate_xt_signature(api_key, api_secret, timestamp, path, query_string="", body_string=""):
    x = "xt-validate-appkey=" + str(api_key) + "&xt-validate-timestamp=" + str(timestamp)
    parts = [path]
    if query_string:
        parts.append(query_string)
    if body_string:
        parts.append(body_string)
    y = "#".join(parts)
    original = x + "#" + y
    return hmac.new(
        api_secret.encode("utf-8"),
        original.encode("utf-8"),
        hashlib.sha256
    ).hexdigest()

class XTClient:

    def __init__(self, api_key, api_secret):
        self.api_key = api_key
        self.api_secret = api_secret
        self.base_url = XT_BASE_URL

    def private_headers(self, timestamp, signature):
        return {
            "Content-Type": "application/x-www-form-urlencoded",
            "xt-validate-appkey": self.api_key,
            "xt-validate-timestamp": timestamp,
            "xt-validate-signature": signature,
            "xt-validate-recvwindow": XT_RECV_WINDOW,
            "xt-validate-algorithms": XT_ALGORITHM,
            "xt-validate-signversion": XT_SIGN_VERSION
        }

    def public_get(self, path, params=None):
        response = http.get(self.base_url + path, params=params or {}, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        return data

    def private_get(self, path, params=None):
        params = params or {}
        timestamp = str(int(time.time() * 1000))
        query_string = canonical_form(params)
        signature = calculate_xt_signature(
            self.api_key,
            self.api_secret,
            timestamp,
            path,
            query_string=query_string,
            body_string=""
        )
        headers = self.private_headers(timestamp, signature)
        response = http.get(self.base_url + path, params=params, headers=headers, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        self.check_api_error(data)
        return data

    def private_post(self, path, params=None):
        params = params or {}
        timestamp = str(int(time.time() * 1000))
        body_string = canonical_form(params)
        signature = calculate_xt_signature(
            self.api_key,
            self.api_secret,
            timestamp,
            path,
            query_string="",
            body_string=body_string
        )
        headers = self.private_headers(timestamp, signature)
        response = http.post(self.base_url + path, data=body_string, headers=headers, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        self.check_api_error(data)
        return data

    @staticmethod
    def check_api_error(data):
        if not isinstance(data, dict):
            return
        rc = data.get("returnCode")
        if rc not in (None, 0, "0"):
            raise RuntimeError(f"XT API error: {data}")

        return self.public_get("/future/market/v1/public/symbol/detail", {"symbol": XT_SYMBOL})

    def normalize_price(self, price):
        if price is None:
            raise ValueError("Price cannot be None.")
        price_decimal = Decimal(str(price))
        if price_decimal <= 0:
            raise ValueError(f"Invalid price: {price}")

        response = self.get_symbol_detail()
        if not isinstance(response, dict):
            raise RuntimeError(f"Invalid XT symbol detail response: {response}")

        result = response.get("result")
        symbol_info = None

        if isinstance(result, list):
            for item in result:
                if not isinstance(item, dict):
                    continue
                if str(item.get("symbol", "")).lower() == XT_SYMBOL.lower():
                    symbol_info = item
                    break
        elif isinstance(result, dict):
            symbol_info = result

        if not isinstance(symbol_info, dict):
            raise RuntimeError(f"XT symbol detail missing for {XT_SYMBOL}: {response}")

        price_precision = symbol_info.get("pricePrecision")
        if price_precision is None:
            raise RuntimeError(f"XT pricePrecision missing for {XT_SYMBOL}: {symbol_info}")

        price_precision = int(price_precision)
        quantizer = Decimal("1").scaleb(-price_precision)
        normalized_price = price_decimal.quantize(quantizer, rounding=ROUND_DOWN)
        normalized_price_xt = format(normalized_price, "f")

        log.warning(
            "XT PRICE NORMALIZED: input=%s precision=%s normalized=%s",
            price, price_precision, normalized_price_xt
        )

        return normalized_price_xt

    def get_mark_price(self):
        return self.public_get("/future/market/v1/public/q/symbol-mark-price", {"symbol": XT_SYMBOL})

    def create_market_entry(self, direction, quantity):
        side = "BUY" if direction == "LONG" else "SELL"
        quantity_decimal = Decimal(str(quantity))
        if quantity_decimal == quantity_decimal.to_integral_value():
            quantity_xt = str(int(quantity_decimal))
        else:
            quantity_xt = format(quantity_decimal, "f").rstrip("0").rstrip(".")

        params = {
            "symbol": XT_SYMBOL,
            "orderSide": side,
            "orderType": "MARKET",
            "origQty": quantity_xt,
            "timeInForce": "IOC",
            "positionSide": direction
        }

        log.warning("REAL XT MARKET ORDER REQUEST: side=%s quantity=%s", side, quantity_xt)
        return self.private_post("/future/trade/v1/order/create", params)

    def create_stop_market(self, direction, quantity, stop_price):
        side = "SELL" if direction == "LONG" else "BUY"
        quantity_decimal = Decimal(str(quantity))
        if quantity_decimal == quantity_decimal.to_integral_value():
            quantity_xt = str(int(quantity_decimal))
        else:
            quantity_xt = format(quantity_decimal, "f").rstrip("0").rstrip(".")

        try:
            mark_response = self.get_mark_price()
            current_xt_price = None
            if isinstance(mark_response, dict):
                result = mark_response.get("result")
                if isinstance(result, dict):
                    current_xt_price = result.get("markPrice") or result.get("price") or result.get("lastPrice")
                elif isinstance(result, list) and result:
                    item = result[0]
                    if isinstance(item, dict):
                        current_xt_price = item.get("markPrice") or item.get("price") or item.get("lastPrice")
            if current_xt_price is not None:
                current_xt_price = float(current_xt_price)
        except Exception as exc:
            log.warning("XT MARK PRICE READ FAILED BEFORE STOP: %s", exc)
            current_xt_price = None

        if current_xt_price is not None:
            if direction == "LONG" and stop_price >= current_xt_price:
                raise RuntimeError(f"INVALID LONG STOP: stop={stop_price} current_xt={current_xt_price}")
            if direction == "SHORT" and stop_price <= current_xt_price:
                raise RuntimeError(f"INVALID SHORT STOP: stop={stop_price} current_xt={current_xt_price}")

        normalized_stop_price = self.normalize_price(stop_price)
        params = {
            "symbol": XT_SYMBOL,
            "orderSide": side,
            "entrustType": "STOP_MARKET",
            "origQty": quantity_xt,
            "stopPrice": normalized_stop_price,
            "timeInForce": "IOC",
            "triggerPriceType": "LATEST_PRICE",
            "positionSide": direction
        }

        log.warning("XT STOP MARKET REQUEST: direction=%s side=%s quantity=%s stopPrice=%s",
                   direction, side, quantity_xt, normalized_stop_price)
        return self.private_post("/future/trade/v1/entrust/create-plan", params)

            quantity_xt = format(quantity_decimal, "f").rstrip("0").rstrip(".")
        
        normalized_tp_price = self.normalize_price(tp_price)
        
        try:

                elif isinstance(result, list) and result:

            if current_price is not None:

def extract_order_id(response):
    if not isinstance(response, dict):
        return None

    for key in ["orderId", "orderID", "id", "entrustId", "entrustID", "order_id"]:
        if key in response and response[key] is not None:
            return str(response[key])

    result = response.get("result")
    if result is not None:
        if isinstance(result, (str, int)):
            return str(result)
        if isinstance(result, dict):
            for key in ["orderId", "orderID", "id", "entrustId", "entrustID", "order_id"]:
                if key in result and result[key] is not None:
                    return str(result[key])
            if "list" in result and isinstance(result["list"], list) and result["list"]:
                first = result["list"][0]
                if isinstance(first, dict):
                    for key in ["orderId", "orderID", "id", "entrustId", "entrustID"]:
                        if key in first and first[key] is not None:
                            return str(first[key])

    return None

# ذخیره کندل‌های دریافتی از ربات (برای تحلیل قدرت)
primary_candles = []
candles_lock = threading.Lock()


def set_primary_candles(candles):
    """دریافت کندل‌ها از ربات (از طریق وب‌هوک یا مستقیم)"""
    global primary_candles
    with candles_lock:
        primary_candles = candles


def get_current_candle():
    """دریافت کندل جاری"""
    with candles_lock:
        if primary_candles:
            return primary_candles[-1]
    return None


def calculate_rsi(candles, period=6):
    """محاسبه RSI از کندل‌های بسته شده"""
    if len(candles) < period + 1:
        return None

    recent = candles[-(period + 1):]
    
    gains = []
    losses = []
    
    for i in range(1, len(recent)):
        change = recent[i]['close'] - recent[i-1]['close']
        if change >= 0:
            gains.append(change)
            losses.append(0)
        else:
            gains.append(0)
            losses.append(abs(change))
    
    if not gains or not losses:
        return None
    
    avg_gain = sum(gains) / len(gains)
    avg_loss = sum(losses) / len(losses)
    
    if avg_loss == 0:
        return 100.0
    
    rs = avg_gain / avg_loss
    rsi = 100 - (100 / (1 + rs))
    
    return rsi


def check_candle_strength(candle, direction):
    """
    بررسی قدرت کندل برای تشخیص ادامه یا برگشت
    
    Returns:
        True: کندل قوی است → 70% TP1، 30% TP2
        False: کندل ضعیف است → 100% در TP1 بسته شود
    """
    if candle is None:
        return True
    
    if not SMART_TP1_ENABLED:
        return True
    
    body = abs(candle["close"] - candle["open"])
    total_range = candle["high"] - candle["low"]
    
    if total_range == 0:
        body_percent = 0
    else:
        body_percent = (body / total_range) * 100
    
    is_bullish = candle["close"] > candle["open"]
    
    # ============================================================
    # 1) بررسی بدنه کندل
    # ============================================================
    if direction == "LONG":
        body_strong = is_bullish and body_percent > SMART_TP1_STRONG_BODY_THRESHOLD
        body_weak = not is_bullish or body_percent < SMART_TP1_WEAK_BODY_THRESHOLD
    else:  # SHORT
        body_strong = not is_bullish and body_percent > SMART_TP1_STRONG_BODY_THRESHOLD
        body_weak = is_bullish or body_percent < SMART_TP1_WEAK_BODY_THRESHOLD
    
    # ============================================================
    # 2) بررسی RSI
    # ============================================================
    rsi = calculate_rsi(primary_candles, 6)
    rsi_ok = False
    if direction == "LONG":
        rsi_ok = rsi is not None and rsi > SMART_TP1_RSI_LONG_THRESHOLD
    else:
        rsi_ok = rsi is not None and rsi < SMART_TP1_RSI_SHORT_THRESHOLD
    
    # ============================================================
    # 3) بررسی حجم
    # ============================================================
    volume_ok = False
    if len(primary_candles) >= 6:
        last_5_volumes = [c.get("samples", 0) for c in primary_candles[-6:-1]]
        avg_volume = sum(last_5_volumes) / len(last_5_volumes) if last_5_volumes else 1
        current_volume = candle.get("samples", 0)
        volume_ok = current_volume > avg_volume * SMART_TP1_VOLUME_MULTIPLIER
    
    # ============================================================
    # 4) تصمیم نهایی
    # ============================================================
    is_strong = body_strong and rsi_ok and volume_ok
    is_weak = body_weak or (not rsi_ok and not volume_ok)
    
    log.info(f"📊 SMART TP1: body={body_percent:.1f}%, rsi={rsi if rsi is None else f'{rsi:.1f}'}, "
             f"volume_ok={volume_ok}, is_strong={is_strong}, is_weak={is_weak}")
    
    return is_strong

def verify_open_position_after_entry(direction, retries=15, delay=0.5):
    last_error = None
    for attempt in range(1, retries + 1):
        try:
            is_open, actual = position_still_open(direction)
            if is_open and actual:
                log.warning("XT POSITION VERIFIED AFTER ENTRY: attempt=%s direction=%s position=%s",
                           attempt, direction, actual)
                return actual
            log.warning("XT POSITION NOT VISIBLE YET: attempt=%s/%s direction=%s",
                       attempt, retries, direction)
        except Exception as exc:
            last_error = exc
            log.warning("XT POSITION VERIFY RETRY: attempt=%s/%s direction=%s error=%s",
                       attempt, retries, direction, exc)

        if attempt < retries:
            time.sleep(delay)

    if last_error is not None:
        raise RuntimeError(f"XT POSITION VERIFY FAILED: {last_error}")
    raise RuntimeError(f"XT POSITION VERIFY FAILED: Position was not returned by XT after {retries} attempts.")

                "quantity": quantity,
                "entry": entry,
                "exit_price": close_price,
                "pnl": pnl,
                "reason": "MANUAL_CLOSE"

                    if not is_strong and (direction == "LONG" and market_price >= tp1_price) or \
                       (direction == "SHORT" and market_price <= tp1_price):
                        
                        log.info(f"⚠️ WEAK CANDLE at TP1 - CLOSING FULL POSITION at {market_price:.4f}")
                        
                        cancel_protection_orders(stop_order_id, tp1_order_id, tp2_order_id)
                        
                        close_response = xt.close_position_market(direction, total_quantity)
                        close_order_id = extract_order_id(close_response)
                        
                        if direction == "LONG":
                            pnl = (market_price - entry) * total_quantity
                        else:
                            pnl = (entry - market_price) * total_quantity
                        
                        remove_active_position(signal_id)
                        
                        append_jsonl(EXECUTION_LOG_FILE, {
                            "time": iso_now(),
                            "event": "POSITION_CLOSED_FULL_TP1_WEAK",
                            "signal_id": signal_id,
                            "direction": direction,
                            "quantity": total_quantity,
                            "entry": entry,
                            "exit_price": market_price,
                            "pnl": pnl,
                            "r": TP_PARTIAL_R,
                            "reason": "WEAK_CANDLE"
                        })
                        
                        send_webhook_to_bot("POSITION_CLOSED", {
                            "signal_id": signal_id,
                            "direction": direction,
                            "entry": entry,
                            "exit_price": market_price,
                            "pnl": pnl,
                            "reason": "FULL_TP1_WEAK_CANDLE",

                        })

            # ============================================================
            # ✅ اگر قدرت دارد → TP1 را فعال کن (70% بسته شود)
            # ============================================================
            if (direction == "LONG" and market_price >= tp1_price) or \
               (direction == "SHORT" and market_price <= tp1_price):

                remaining_quantity = total_quantity - tp1_quantity

                    direction,
                    remaining_quantity,
                    stop_order_id,
                    tp1_order_id,
                    tp2_order_id,

                    tp2_price
                )

                    "stop_order_id": new_stop_order_id,
                    "tp1_order_id": None,

                    "tp1_closed": True,

                    "time": iso_now(),
                    "event": "POSITION_BREAKEVEN_ACTIVATED",
                    "signal_id": signal_id,
                    "direction": direction,
                    "entry": entry,
                    "risk_R": risk_r,
                    "tp1_price": tp1_price,
                    "remaining_quantity": remaining_quantity,
                    "stop_order_id": new_stop_order_id,

                    "signal_id": signal_id,
                    "direction": direction,
                    "entry": entry,

                })

        # ============================================================
        if stage == 1 and position_data.get("tp1_closed", False):
            
            remaining_quantity = total_quantity - tp1_quantity
            
            # ------------------------------------------------
            # ✅ اگر قیمت به Breakeven برگشت → کل پوزیشن بسته شود
            # ------------------------------------------------
            if (direction == "LONG" and market_price <= entry) or \
               (direction == "SHORT" and market_price >= entry):
                
                log.info(f"⚠️ Position {signal_id} returned to breakeven - CLOSING REMAINING {remaining_quantity:.6f}")
                
                cancel_protection_orders(stop_order_id, None, tp2_order_id)

                    pnl = (tp1_price - entry) * tp1_quantity
                else:
                    pnl = (entry - tp1_price) * tp1_quantity

            # ------------------------------------------------

                    "event": "POSITION_CLOSED_FULL_TP",
                    "signal_id": signal_id,
                    "direction": direction,
                    "entry": entry,

                    "tp2_price": tp2_price,

                    "pnl": pnl
                })

                    "signal_id": signal_id,
                    "direction": direction,
                    "entry_price": entry,
                    "exit_price": tp2_price,
                    "pnl": pnl,
                    "reason": "TP2_FULL_PROFIT",
                    "total_r": TP_FULL_R

def get_order_quantity():
    detail = xt.get_symbol_detail()
    result = detail.get("result", {}) if isinstance(detail, dict) else {}

    contract_size = float(result.get("contractSize", 0) or 0)
    if contract_size <= 0:
        raise RuntimeError("XT contractSize unavailable.")

    mark_data = xt.get_mark_price()
    mark_result = mark_data.get("result", {}) if isinstance(mark_data, dict) else {}

    mark_price = float(mark_result.get("p", 0) or 0)
    if mark_price <= 0:
        raise RuntimeError("XT mark price unavailable.")

    margin = float(config["margin_usdt"])
    leverage = int(config["leverage"])

    raw_quantity = margin * leverage / (mark_price * contract_size)

    step = None
    candidates = [
        result.get("quantityPrecision"),
        result.get("basePrecision"),
        result.get("stepSize"),
        result.get("minQty")
    ]

    for candidate in candidates:
        if isinstance(candidate, (int, float)):
            value = float(candidate)
            if 0 < value < 1:
                step = value
                break

    if step is None:
        for candidate in candidates:
            if candidate is None:
                continue
            try:
                value = Decimal(str(candidate))
                if 0 < value < 1:
                    step = value
                    break
            except Exception:
                pass

    if step is None:
        precision = result.get("quantityPrecision")
        if isinstance(precision, int) and precision >= 0:
            step = 10 ** (-precision)

    if step is None:
        step = 1.0

    raw_decimal = Decimal(str(raw_quantity))
    step_decimal = Decimal(str(step))

    if step_decimal <= 0:
        raise RuntimeError("Invalid XT quantity step.")

    units = raw_decimal / step_decimal
    units_up = units.to_integral_value(rounding=ROUND_UP)
    quantity_decimal = units_up * step_decimal

    if quantity_decimal <= 0:
        raise RuntimeError(f"Calculated XT order quantity is zero. raw_quantity={raw_quantity}, step={step_decimal}")

    quantity = float(quantity_decimal)
    return {
        "quantity": quantity,
        "raw_quantity": raw_quantity,
        "mark_price": mark_price,
        "contract_size": contract_size,
        "quantity_step": float(step_decimal),
        "margin": margin,
        "leverage": leverage
    }

def execute_signal(signal):
    signal_id = signal["signal_id"]
    direction = signal["direction"]

    with state_lock:
        already_consumed = state.get("signal_consumed") is True and state.get("last_signal_id") == signal_id

    if already_consumed:
        log.warning("SIGNAL ALREADY CONSUMED - ENTRY BLOCKED: %s", signal_id)
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "SIGNAL_ALREADY_CONSUMED"
        }

    existing_position = get_active_position(signal_id)
    if existing_position:
        log.warning(f"Position already exists for signal_id: {signal_id}")
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "POSITION_ALREADY_EXISTS"
        }

    with state_lock:
        if state.get("last_signal_id") == signal_id and state.get("entry_confirmed"):
            return {
                "accepted": False,
                "status": "DUPLICATE_SIGNAL",
                "signal_id": signal_id
            }

    # ============================================================
    # ✅ بررسی: آیا پوزیشن برای این Symbol باز است؟ (هر جهتی)
    # ============================================================
    has_position, position = has_open_position_for_symbol()
    if has_position:
        log.warning(f"❌ OPEN POSITION EXISTS for {DISPLAY_SYMBOL} - SIGNAL REJECTED: {signal_id}")
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "POSITION_ALREADY_EXISTS",
            "position": position,
            "symbol": DISPLAY_SYMBOL
        }

    # ============================================================
    # ✅ بررسی: آیا سفارش باز برای این Symbol وجود دارد؟
    # ============================================================
    has_order, order = has_open_orders_for_symbol()
    if has_order:
        log.warning(f"❌ OPEN ORDER EXISTS for {DISPLAY_SYMBOL} - SIGNAL REJECTED: {signal_id}")
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "OPEN_ORDER_EXISTS",
            "order": order,
            "symbol": DISPLAY_SYMBOL
        }

    if not TRADING_ENABLED:
        return {
            "accepted": False,
            "status": "TRADING_DISABLED",
            "signal_id": signal_id
        }

    if not ORDERS_ENABLED:
        return {
            "accepted": False,
            "status": "ORDERS_DISABLED",
            "signal_id": signal_id
        }

    display_signal(signal)

    append_jsonl(SIGNAL_LOG_FILE, {
        "received_at": iso_now(),
        "signal": signal
    })

    send_webhook_to_bot("SIGNAL_RECEIVED", {
        "signal_id": signal_id,
        "direction": direction,
        "reference_entry": signal.get("reference_entry"),
        "initial_risk": signal.get("initial_risk")
    })

    with state_lock:
        state["connected_to_bot"] = True
        state["last_signal_time"] = iso_now()
        state["execution_status"] = "SIGNAL_RECEIVED"

    save_state()

    if not LIVE_ORDER_EXECUTION:
        with state_lock:
            state["execution_status"] = "LIVE_EXECUTION_DISABLED"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "LIVE_EXECUTION_DISABLED"
        }

    if not state.get("xt_private_ready"):
        with state_lock:
            state["execution_status"] = "SIGNAL_RECEIVED_XT_PRIVATE_PARTIAL"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "SIGNAL_RECEIVED_XT_PRIVATE_PARTIAL"
        }

    try:
        balance_response = xt.get_balance()
        available = extract_usdt_balance(balance_response)
        with state_lock:
            state["xt_balance"] = available
            state["xt_balance_status"] = "OK"
        save_state()
    except Exception as exc:
        with state_lock:
            state["execution_status"] = "BALANCE_READ_FAILED"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "BALANCE_READ_FAILED",
            "error": str(exc)
        }

    try:
        quantity_info = get_order_quantity()
    except Exception as exc:
        with state_lock:
            state["execution_status"] = "QUANTITY_CALCULATION_FAILED"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "QUANTITY_CALCULATION_FAILED",
            "error": str(exc)
        }

    total_quantity = quantity_info["quantity"]

    if total_quantity <= 0:
        with state_lock:
            state["execution_status"] = "INVALID_ORDER_QUANTITY"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "INVALID_ORDER_QUANTITY"
        }

    step = quantity_info["quantity_step"]
    tp1_quantity = total_quantity * TP_PARTIAL_PERCENT
    tp1_quantity_decimal = Decimal(str(tp1_quantity))
    step_decimal = Decimal(str(step))
    units = tp1_quantity_decimal / step_decimal
    tp1_quantity = float(units.to_integral_value(rounding=ROUND_DOWN) * step_decimal)
    
    if tp1_quantity >= total_quantity:
        tp1_quantity = total_quantity * 0.5
    
    tp2_quantity = total_quantity - tp1_quantity

    print()
    print("=" * 78)
    print(f"XT LIVE EXECUTION PREPARATION ({DISPLAY_SYMBOL})")
    print("=" * 78)
    print(f"AVAILABLE USDT   : {available}")
    print(f"MARGIN           : {quantity_info['margin']}")
    print(f"LEVERAGE         : {quantity_info['leverage']}x")
    print(f"MARK PRICE       : {quantity_info['mark_price']:.4f}")
    print(f"TOTAL QTY        : {total_quantity}")
    print(f"STOP             : {STOP_R_MULTIPLE}R (Fixed)")
    print(f"TP1 QTY (70%)    : {tp1_quantity} @ {TP_PARTIAL_R}R")
    print(f"TP2 QTY (30%)    : {tp2_quantity} @ {TP_FULL_R}R")
    print(f"BREAKEVEN        : {BREAKEVEN_TRIGGER_R}R (Stop -> Entry)")
    print(f"SMART TP1        : {'ENABLED' if SMART_TP1_ENABLED else 'DISABLED'}")
    print("  - کندل قوی → 70% TP1, 30% TP2")
    print("  - کندل ضعیف → 100% TP1")
    print("✅ OPEN POSITION CHECK : NONE")
    print("✅ OPEN ORDER CHECK    : NONE")
    print("LIVE ORDER       : TRUE")
    print("=" * 78)

    if available is not None and available <= 0:
        print()
        print("⚠️ XT AVAILABLE BALANCE = 0.0 USDT")
        print("REAL MARKET ORDER WILL BE SENT.")

    with state_lock:
        state["execution_status"] = "ENTRY_SUBMITTING"
    save_state()

    r_value = float(signal["initial_risk"])
    entry = float(signal["reference_entry"])
    
    final_stop = entry - (r_value * STOP_R_MULTIPLE) if direction == "LONG" else entry + (r_value * STOP_R_MULTIPLE)
    tp1_price = entry + (r_value * TP_PARTIAL_R) if direction == "LONG" else entry - (r_value * TP_PARTIAL_R)
    tp2_price = entry + (r_value * TP_FULL_R) if direction == "LONG" else entry - (r_value * TP_FULL_R)

    if final_stop <= 0:
        raise RuntimeError("Calculated stop price is invalid.")

    if tp1_price <= 0 or tp2_price <= 0:
        raise RuntimeError("Calculated TP price is invalid.")

    if direction == "LONG":
        if final_stop >= entry:
            raise RuntimeError(f"LONG stop ({final_stop:.4f}) is not below entry ({entry:.4f}).")
        if tp1_price <= entry or tp2_price <= entry:
            raise RuntimeError(f"LONG TP ({tp1_price:.4f}, {tp2_price:.4f}) is not above entry ({entry:.4f}).")
    else:
        if final_stop <= entry:
            raise RuntimeError(f"SHORT stop ({final_stop:.4f}) is not above entry ({entry:.4f}).")
        if tp1_price >= entry or tp2_price >= entry:
            raise RuntimeError(f"SHORT TP ({tp1_price:.4f}, {tp2_price:.4f}) is not below entry ({entry:.4f}).")

    print()
    print("📊 CALCULATED PRICES:")
    print(f"   ENTRY : {entry:.4f}")
    print(f"   STOP  : {final_stop:.4f} ({STOP_R_MULTIPLE}R)")
    print(f"   TP1   : {tp1_price:.4f} ({TP_PARTIAL_R}R)")
    print(f"   TP2   : {tp2_price:.4f} ({TP_FULL_R}R)")

    try:
        print()
        print("🔥 SENDING REAL XT MARKET ENTRY...")

        entry_response = xt.create_market_entry(direction, total_quantity)
        entry_order_id = extract_order_id(entry_response)

        log.warning(
            "REAL_ENTRY_RESPONSE: signal_id=%s direction=%s quantity=%s order_id=%s response=%s",
            signal_id, direction, total_quantity, entry_order_id, entry_response
        )

        if entry_order_id is None:
            raise RuntimeError(
                f"XT MARKET ENTRY succeeded but order ID could not be extracted: {entry_response}"
            )

        with state_lock:
            state["last_signal_id"] = signal_id
            state["last_signal_time"] = iso_now()
            state["entry_confirmed"] = True
            state["entry_retry_allowed"] = False
            state["signal_consumed"] = True
            state["execution_status"] = "ENTRY_CONFIRMED"

        save_state()

        append_jsonl(EXECUTION_LOG_FILE, {
            "time": iso_now(),
            "event": "SIGNAL_CONSUMED_AFTER_ENTRY",
            "signal_id": signal_id,
            "direction": direction,
            "entry_order_id": entry_order_id

class SignalHandler(BaseHTTPRequestHandler):

    def log_message(self, format, *args):
        return

    def send_json(self, status_code, payload):
        try:
            body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
            self.send_response(status_code)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
        except (BrokenPipeError, ConnectionResetError):
            log.warning("HTTP CLIENT DISCONNECTED BEFORE RESPONSE COULD BE SENT.")
        except Exception:
            log.exception("HTTP JSON RESPONSE SEND FAILED")

    def do_GET(self):
        if self.path == "/health":
            with state_lock:
                positions = get_all_active_positions()
                payload = {
                    "platform": "REZA EXECUTION PLATFORM",
                    "version": VERSION,
                    "bot_version": BOT_VERSION,
                    "symbol": DISPLAY_SYMBOL,
                    "status": "ONLINE",
                    "live_order_execution": LIVE_ORDER_EXECUTION,
                    "orders_enabled": ORDERS_ENABLED,
                    "trading_enabled": TRADING_ENABLED,
                    "webhook_enabled": BOT_WEBHOOK_ENABLED,
                    "bot_connection": state.get("connected_to_bot"),
                    "xt_private_ready": state.get("xt_private_ready"),
                    "xt_account_readable": state.get("xt_account_readable"),
                    "xt_balance": state.get("xt_balance"),
                    "xt_balance_status": state.get("xt_balance_status"),
                    "xt_position_status": state.get("xt_position_status"),
                    "xt_order_api_status": state.get("xt_order_api_status"),
                    "last_signal_id": state.get("last_signal_id"),
                    "execution_status": state.get("execution_status"),
                    "active_positions_count": len(positions),
                    "active_positions": positions,
                    "management_mode": "SMART_TP1_XRP_V1",
                    "smart_tp1_enabled": SMART_TP1_ENABLED,
                    "stop_r": STOP_R_MULTIPLE,
                    "tp1_r": TP_PARTIAL_R,
                    "tp2_r": TP_FULL_R,
                    "tp1_percent": TP_PARTIAL_PERCENT,
                    "breakeven_trigger_r": BREAKEVEN_TRIGGER_R,
                    "open_position_check": True,
                    "open_order_check": True
                }
            self.send_json(200, payload)
            return

        self.send_json(404, {"error": "NOT_FOUND"})

    def do_POST(self):
        if self.path != BOT_SIGNAL_PATH:
            self.send_json(404, {"accepted": False, "error": "NOT_FOUND"})
            return

        try:
            length = int(self.headers.get("Content-Length", "0"))
            raw = self.rfile.read(length)
            signal = json.loads(raw.decode("utf-8"))
            validate_signal(signal)
            result = execute_signal(signal)
            self.send_json(200, result)
        except Exception as exc:
            log.exception("Signal processing failed")
            self.send_json(400, {
                "accepted": False,
                "status": "REJECTED",
                "error": str(exc)
            })

server = None


class ReusableThreadingTCPServer(ThreadingTCPServer):
    allow_reuse_address = True
    daemon_threads = True


def start_signal_server():
    global server

    try:
        server = ReusableThreadingTCPServer(
            (PLATFORM_HOST, PLATFORM_PORT),
            SignalHandler
        )
    except OSError as exc:
        if getattr(exc, "errno", None) == 98:
            print()
            print("=" * 78)
            print("BOT SIGNAL SERVER")
            print("=" * 78)
            print(f"PORT {PLATFORM_PORT} IS ALREADY IN USE.")
            print("NO SECOND LISTENER WILL BE CREATED.")
            print("Existing REZA Execution Platform instance may already be active.")
            print("=" * 78)

            with state_lock:
                state["listener_owner"] = "EXISTING_INSTANCE"
            save_state()
            return False
        raise

    thread = threading.Thread(
        target=server.serve_forever,
        daemon=True,
        name="REZA-PLATFORM-LISTENER-XRP-SMART"
    )
    thread.start()

    with state_lock:
        state["listener_owner"] = f"{PLATFORM_HOST}:{PLATFORM_PORT}"
    save_state()

    print()
    print("=" * 78)
    print("BOT SIGNAL SERVER")
    print("=" * 78)
    print("LISTENER           : RUNNING")
    print(f"SIGNAL             : http://{PLATFORM_HOST}:{PLATFORM_PORT}{BOT_SIGNAL_PATH}")
    print(f"HEALTH             : http://{PLATFORM_HOST}:{PLATFORM_PORT}/health")
    print("=" * 78)

    return True
