# XT PARAMETER CANONICALIZATION
# ============================================================================

def canonical_form(params):
    if not params:
        return ""
    pairs = []
    for key in sorted(params.keys()):
        value = params[key]
        if value is None:
            continue
        if isinstance(value, bool):
            value = "true" if value else "false"
        pairs.append(f"{key}={value}")
    return "&".join(pairs)


# ============================================================================
# XT SIGNATURE
# ============================================================================

def calculate_xt_signature(api_key, api_secret, timestamp, path, query_string="", body_string=""):
    x = "xt-validate-appkey=" + str(api_key) + "&xt-validate-timestamp=" + str(timestamp)
    parts = [path]
    if query_string:
        parts.append(query_string)
    if body_string:
        parts.append(body_string)
    y = "#".join(parts)
    original = x + "#" + y
    return hmac.new(
        api_secret.encode("utf-8"),
        original.encode("utf-8"),
        hashlib.sha256
    ).hexdigest()

class XTClient:

    def __init__(self, api_key, api_secret):
        self.api_key = api_key
        self.api_secret = api_secret
        self.base_url = XT_BASE_URL

    def private_headers(self, timestamp, signature):
        return {
            "Content-Type": "application/x-www-form-urlencoded",
            "xt-validate-appkey": self.api_key,
            "xt-validate-timestamp": timestamp,
            "xt-validate-signature": signature,
            "xt-validate-recvwindow": XT_RECV_WINDOW,
            "xt-validate-algorithms": XT_ALGORITHM,
            "xt-validate-signversion": XT_SIGN_VERSION
        }

    def public_get(self, path, params=None):
        response = http.get(self.base_url + path, params=params or {}, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        return data

    def private_get(self, path, params=None):
        params = params or {}
        timestamp = str(int(time.time() * 1000))
        query_string = canonical_form(params)
        signature = calculate_xt_signature(
            self.api_key,
            self.api_secret,
            timestamp,
            path,
            query_string=query_string,
            body_string=""
        )
        headers = self.private_headers(timestamp, signature)
        response = http.get(self.base_url + path, params=params, headers=headers, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        self.check_api_error(data)
        return data

    def private_post(self, path, params=None):
        params = params or {}
        timestamp = str(int(time.time() * 1000))
        body_string = canonical_form(params)
        signature = calculate_xt_signature(
            self.api_key,
            self.api_secret,
            timestamp,
            path,
            query_string="",
            body_string=body_string
        )
        headers = self.private_headers(timestamp, signature)
        response = http.post(self.base_url + path, data=body_string, headers=headers, timeout=10)
        try:
            data = response.json()
        except Exception:
            data = response.text
        if response.status_code >= 400:
            raise RuntimeError(f"XT HTTP {response.status_code}: {data}")
        self.check_api_error(data)
        return data

    @staticmethod
    def check_api_error(data):
        if not isinstance(data, dict):
            return
        rc = data.get("returnCode")
        if rc not in (None, 0, "0"):
            raise RuntimeError(f"XT API error: {data}")

    def normalize_price(self, price):
        if price is None:
            raise ValueError("Price cannot be None.")
        price_decimal = Decimal(str(price))
        if price_decimal <= 0:
            raise ValueError(f"Invalid price: {price}")

        response = self.get_symbol_detail()
        if not isinstance(response, dict):
            raise RuntimeError(f"Invalid XT symbol detail response: {response}")

        result = response.get("result")
        symbol_info = None

        if isinstance(result, list):
            for item in result:
                if not isinstance(item, dict):
                    continue
                if str(item.get("symbol", "")).lower() == XT_SYMBOL.lower():
                    symbol_info = item
                    break
        elif isinstance(result, dict):
            symbol_info = result

        if not isinstance(symbol_info, dict):
            raise RuntimeError(f"XT symbol detail missing for {XT_SYMBOL}: {response}")

        price_precision = symbol_info.get("pricePrecision")
        if price_precision is None:
            raise RuntimeError(f"XT pricePrecision missing for {XT_SYMBOL}: {symbol_info}")

        price_precision = int(price_precision)
        quantizer = Decimal("1").scaleb(-price_precision)
        normalized_price = price_decimal.quantize(quantizer, rounding=ROUND_DOWN)
        normalized_price_xt = format(normalized_price, "f")

        log.warning(
            "XT PRICE NORMALIZED: input=%s precision=%s normalized=%s",
            price, price_precision, normalized_price_xt
        )

        return normalized_price_xt

    def get_quantity_step(self):
        detail = self.get_symbol_detail()
        result = detail.get("result", {}) if isinstance(detail, dict) else {}

        step = None
        candidates = [
            result.get("quantityPrecision"),
            result.get("basePrecision"),
            result.get("stepSize"),
            result.get("minQty")
        ]

        for candidate in candidates:
            if isinstance(candidate, (int, float)):
                value = float(candidate)
                if 0 < value < 1:
                    step = value
                    break

        if step is None:
            for candidate in candidates:
                if candidate is None:
                    continue
                try:
                    value = Decimal(str(candidate))
                    if 0 < value < 1:
                        step = value
                        break
                except Exception:
                    pass

        if step is None:
            precision = result.get("quantityPrecision")
            if isinstance(precision, int) and precision >= 0:
                step = 10 ** (-precision)

        if step is None:
            step = 1.0

        return float(step)

        return float(price)

    def create_stop_market(self, direction, quantity, stop_price):
        side = "SELL" if direction == "LONG" else "BUY"
        quantity_decimal = Decimal(str(quantity))
        if quantity_decimal == quantity_decimal.to_integral_value():
            quantity_xt = str(int(quantity_decimal))
        else:
            quantity_xt = format(quantity_decimal, "f").rstrip("0").rstrip(".")

        try:
            mark_response = self.get_mark_price()
            current_xt_price = None
            if isinstance(mark_response, dict):
                result = mark_response.get("result")
                if isinstance(result, dict):
                    current_xt_price = result.get("markPrice") or result.get("price") or result.get("lastPrice")
                elif isinstance(result, list) and result:
                    item = result[0]
                    if isinstance(item, dict):
                        current_xt_price = item.get("markPrice") or item.get("price") or item.get("lastPrice")
            if current_xt_price is not None:
                current_xt_price = float(current_xt_price)
        except Exception as exc:
            log.warning("XT MARK PRICE READ FAILED BEFORE STOP: %s", exc)
            current_xt_price = None

        if current_xt_price is not None:
            if direction == "LONG" and stop_price >= current_xt_price:
                raise RuntimeError(f"INVALID LONG STOP: stop={stop_price} current_xt={current_xt_price}")
            if direction == "SHORT" and stop_price <= current_xt_price:
                raise RuntimeError(f"INVALID SHORT STOP: stop={stop_price} current_xt={current_xt_price}")

        normalized_stop_price = self.normalize_price(stop_price)
        params = {
            "symbol": XT_SYMBOL,
            "orderSide": side,
            "entrustType": "STOP_MARKET",
            "origQty": quantity_xt,
            "stopPrice": normalized_stop_price,
            "timeInForce": "IOC",
            "triggerPriceType": "LATEST_PRICE",
            "positionSide": direction
        }

        log.warning("XT STOP MARKET REQUEST: direction=%s side=%s quantity=%s stopPrice=%s",
                   direction, side, quantity_xt, normalized_stop_price)
        return self.private_post("/future/trade/v1/entrust/create-plan", params)

        try:

                elif isinstance(result, list) and result:

            if current_price is not None:

                if direction == "LONG" and tp_price <= current_price:
                    log.warning("LONG TP is not above current price: tp=%s current=%s", tp_price, current_price)
                if direction == "SHORT" and tp_price >= current_price:
                    log.warning("SHORT TP is not below current price: tp=%s current=%s", tp_price, current_price)

        params = {
            "symbol": XT_SYMBOL,
            "orderSide": side,
            "orderType": "LIMIT",
            "origQty": quantity_xt,
            "price": normalized_tp_price,
            "timeInForce": "GTC",
            "positionSide": direction,

        }

        log.warning("XT TAKE PROFIT LIMIT REQUEST: direction=%s side=%s quantity=%s tpPrice=%s reduceOnly=%s",
                   direction, side, quantity_xt, normalized_tp_price, reduce_only)
        return self.private_post("/future/trade/v1/order/create", params)

    def cancel_order(self, order_id):
        if order_id is None:
            return True
        
        try:
            order_id_int = int(order_id)
        except (ValueError, TypeError):
            order_id_int = order_id
            
        response = self.private_post("/future/trade/v1/order/cancel", {"orderId": order_id_int})
        if not isinstance(response, dict):
            raise RuntimeError(f"Invalid cancel response: {response}")
        return response

    def close_position_market(self, direction, quantity):
        direction = str(direction).upper()
        if direction == "LONG":
            order_side = "SELL"
        elif direction == "SHORT":
            order_side = "BUY"
        else:
            raise RuntimeError(f"Invalid direction: {direction}")

        quantity_decimal = Decimal(str(quantity))
        if quantity_decimal == quantity_decimal.to_integral_value():
            quantity_xt = str(int(quantity_decimal))
        else:
            quantity_xt = format(quantity_decimal, "f").rstrip("0").rstrip(".")

        response = self.private_post(
            "/future/trade/v1/order/create",
            {
                "symbol": XT_SYMBOL,
                "orderSide": order_side,
                "orderType": "MARKET",
                "origQty": quantity_xt,
                "reduceOnly": True,
                "positionSide": direction
            }
        )
        if not isinstance(response, dict):
            raise RuntimeError(f"Invalid close response: {response}")
        return response

def extract_order_id(response):
    if not isinstance(response, dict):
        return None

    for key in ["orderId", "orderID", "id", "entrustId", "entrustID", "order_id"]:
        if key in response and response[key] is not None:
            return str(response[key])

    result = response.get("result")
    if result is not None:
        if isinstance(result, (str, int)):
            return str(result)
        if isinstance(result, dict):
            for key in ["orderId", "orderID", "id", "entrustId", "entrustID", "order_id"]:
                if key in result and result[key] is not None:
                    return str(result[key])
            if "list" in result and isinstance(result["list"], list) and result["list"]:
                first = result["list"][0]
                if isinstance(first, dict):
                    for key in ["orderId", "orderID", "id", "entrustId", "entrustID"]:
                        if key in first and first[key] is not None:
                            return str(first[key])

    return None

def extract_positions(response):
    if not isinstance(response, dict):
        return []

    result = response.get("result")
    if result is None:
        return []

    if isinstance(result, list):
        return result

    if isinstance(result, dict):
        positions = result.get("list")
        if isinstance(positions, list):
            return positions
        if "positionSize" in result:
            return [result]

    return []

        if value is None:

def calculate_position_size(entry_price, stop_price, direction):
    if not USE_FIXED_RISK:
        return get_order_quantity_legacy()
    
    if direction == "LONG":
        risk_distance = abs(entry_price - stop_price)
    else:
        risk_distance = abs(stop_price - entry_price)
    
    if risk_distance <= 0:
        raise RuntimeError("Invalid risk distance for position sizing.")
    
    raw_quantity = FIXED_RISK_USDT / risk_distance
    
    step = xt.get_quantity_step()
    quantity_decimal = Decimal(str(raw_quantity))
    step_decimal = Decimal(str(step))
    units = quantity_decimal / step_decimal
    quantity = float(units.to_integral_value(rounding=ROUND_DOWN) * step_decimal)
    
    if quantity <= 0:
        quantity = step
    
    mark_price_value = xt.get_latest_price()
    contract_size = xt.get_contract_size()
    notional_value = quantity * mark_price_value * contract_size
    
    required_margin = FIXED_RISK_USDT
    leverage = notional_value / required_margin if required_margin > 0 else 1
    
    log.info(f"📊 FIXED RISK SIZING: risk=${FIXED_RISK_USDT}, "
             f"stop_distance={risk_distance:.4f}, quantity={quantity:.4f}, "
             f"notional=${notional_value:.2f}, leverage={leverage:.1f}x")
    
    return {
        "quantity": quantity,
        "margin": required_margin,
        "leverage": min(leverage, 50),
        "quantity_step": step,
        "mark_price": mark_price_value,
        "contract_size": contract_size,
        "risk_usdt": FIXED_RISK_USDT,
        "stop_distance": risk_distance
    }


def get_order_quantity_legacy():
    detail = xt.get_symbol_detail()
    result = detail.get("result", {}) if isinstance(detail, dict) else {}

    contract_size = float(result.get("contractSize", 0) or 0)
    if contract_size <= 0:
        raise RuntimeError("XT contractSize unavailable.")

    mark_price = xt.get_latest_price()

    margin = float(config.get("margin_usdt", 4.0))
    leverage = int(config.get("leverage", 8))

    raw_quantity = margin * leverage / (mark_price * contract_size)

    step = xt.get_quantity_step()

    raw_decimal = Decimal(str(raw_quantity))
    step_decimal = Decimal(str(step))

    if step_decimal <= 0:
        raise RuntimeError("Invalid XT quantity step.")

    units = raw_decimal / step_decimal
    units_up = units.to_integral_value(rounding=ROUND_UP)
    quantity_decimal = units_up * step_decimal

    if quantity_decimal <= 0:
        raise RuntimeError(f"Calculated XT order quantity is zero.")

    quantity = float(quantity_decimal)
    return {
        "quantity": quantity,
        "mark_price": mark_price,
        "contract_size": contract_size,
        "quantity_step": float(step_decimal),
        "margin": margin,
        "leverage": leverage,
        "risk_usdt": margin
    }


def get_order_quantity(entry_price=None, stop_price=None, direction=None):
    if USE_FIXED_RISK and entry_price is not None and stop_price is not None:
        return calculate_position_size(entry_price, stop_price, direction)
    return get_order_quantity_legacy()

def cancel_protection_orders(stop_order_id, tp_order_id):
    errors = []
    for order_id in (stop_order_id, tp_order_id):
        if order_id is None:
            continue
        try:
            xt.cancel_order(order_id)
        except Exception as exc:
            errors.append(f"{order_id}: {exc}")

    if errors:
        log.warning(f"Protection cancellation errors: {errors}")


def manage_single_position(signal_id, position_data):
    """
    مدیریت یک پوزیشن با منطق صحیح:
    - بریک‌ایون در 0.5R
    - بسته شدن کامل در 1R (TP)
    - بعد از بسته شدن، پوزیشن از لیست حذف می‌شود
    """
    try:
        direction = str(position_data.get("direction", "")).upper()
        entry = float(position_data.get("entry", 0))
        risk_r = float(position_data.get("risk_R", 0))
        total_quantity = float(position_data.get("quantity", 0))
        stop_order_id = position_data.get("stop_order_id")
        tp_order_id = position_data.get("tp_order_id")
        stage = int(position_data.get("management_stage", 0))
        
        # ✅ محاسبه قیمت‌ها با TP=1R
        tp_price = entry + (risk_r * TP_FULL_R) if direction == "LONG" else entry - (risk_r * TP_FULL_R)
        breakeven_trigger = entry + (risk_r * BREAKEVEN_TRIGGER_R) if direction == "LONG" else entry - (risk_r * BREAKEVEN_TRIGGER_R)

        # ============================================================
        # ✅ STEP 0: بررسی اینکه پوزیشن هنوز باز است
        # ============================================================
        is_open, actual = position_still_open(direction)
        
        if not is_open:
            log.info(f"🔒 Position {signal_id} is already closed. Removing from tracking.")
            remove_active_position(signal_id)
            return

        # دریافت قیمت بازار
        market_price = xt.get_latest_price()

        # ============================================================
        # ✅ STEP 1: بریک‌ایون در 0.5R
        # ============================================================
        if stage == 0:
            if (direction == "LONG" and market_price >= breakeven_trigger) or \
               (direction == "SHORT" and market_price <= breakeven_trigger):
                
                log.info(f"✅ Position {signal_id} reached 0.5R - BREAKEVEN ACTIVATED")
                
                # لغو سفارشات قبلی
                cancel_protection_orders(stop_order_id, tp_order_id)
                
                # ایجاد استاپ جدید روی قیمت ورود
                new_stop_response = xt.create_stop_market(direction, total_quantity, entry)
                new_stop_order_id = extract_order_id(new_stop_response)
                
                position_data.update({
                    "stop_order_id": new_stop_order_id,
                    "stop": entry,
                    "management_stage": 1,
                    "breakeven_trigger_price": breakeven_trigger
                })
                add_active_position(signal_id, position_data)
                return

        # ============================================================
        # ✅ STEP 2: TP کامل (100% در 1R)
        # ============================================================
        if (direction == "LONG" and market_price >= tp_price) or \
           (direction == "SHORT" and market_price <= tp_price):
            
            log.info(f"✅ Position {signal_id} reached TP (1R) - 100% CLOSED")
            
            # ❌ حذف پوزیشن از ترکینگ قبل از هر چیز
            remove_active_position(signal_id)
            
            # ✅ ثبت لاگ و ارسال وب‌هوک
            append_jsonl(EXECUTION_LOG_FILE, {
                "time": iso_now(),
                "event": "POSITION_CLOSED_FULL_TP",
                "signal_id": signal_id,
                "direction": direction,
                "entry": entry,
                "tp_price": tp_price,
                "total_r": TP_FULL_R
            })
            
            send_webhook_to_bot("POSITION_CLOSED", {
                "signal_id": signal_id,
                "direction": direction,
                "entry": entry,
                "exit_price": tp_price,
                "reason": "TP_FULL_PROFIT",
                "total_r": TP_FULL_R
            })
            
            # ✅ تلاش برای لغو سفارشات محافظتی (اختیاری)
            try:
                cancel_protection_orders(stop_order_id, tp_order_id)
            except Exception as exc:
                log.warning(f"Could not cancel protection orders: {exc}")
            
            return

    except Exception as exc:
        log.error(f"Error managing position {signal_id}: {exc}")

    with state_lock:

            "accepted": False,
            "signal_id": signal_id,
            "status": "SIGNAL_ALREADY_CONSUMED"
        }

    existing_position = get_active_position(signal_id)
    if existing_position:
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "POSITION_ALREADY_EXISTS"
        }

    with state_lock:
        if state.get("last_signal_id") == signal_id and state.get("entry_confirmed"):

                "accepted": False,
                "status": "DUPLICATE_SIGNAL",
                "signal_id": signal_id
            }

    has_position, position = has_open_position_for_symbol()
    if has_position:
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "POSITION_ALREADY_EXISTS"

    if has_order:
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "OPEN_ORDER_EXISTS"

    for attempt in range(15):
        try:
            position_response = xt.get_position()

            if actual:
                print(f"XT POSITION CONFIRMED (attempt {attempt+1})")
                break
            time.sleep(0.5)
        except Exception:
            pass

    if actual is None:
        with state_lock:
            state["execution_status"] = "POSITION_VERIFY_FAILED"
        save_state()
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "POSITION_VERIFY_FAILED"
        }

    actual_size = float(actual.get("positionSize", 0) or 0)
    entry_price = float(actual.get("entryPrice", 0) or 0)

    if entry_price <= 0 or actual_size <= 0:
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "INVALID_POSITION"
        }

    # ✅ محاسبه مجدد با قیمت واقعی
    r_value = float(signal["initial_risk"])
    final_stop = entry_price - (r_value * STOP_R_MULTIPLE) if direction == "LONG" else entry_price + (r_value * STOP_R_MULTIPLE)
    tp_price = entry_price + (r_value * TP_FULL_R) if direction == "LONG" else entry_price - (r_value * TP_FULL_R)

    # ✅ ایجاد استاپ (1R)
    stop_response = xt.create_stop_market(direction, actual_size, final_stop)
    stop_order_id = extract_order_id(stop_response)

    if stop_order_id is None:
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "STOP_FAILED"
        }

    # ✅ ایجاد TP (1R - 100% پوزیشن)
    tp_response = xt.create_take_profit_limit(direction, actual_size, tp_price, reduce_only=True)
    tp_order_id = extract_order_id(tp_response)

    if tp_order_id is None:
        try:
            xt.cancel_order(stop_order_id)
        except:
            pass
        return {
            "accepted": False,
            "signal_id": signal_id,
            "status": "TP_FAILED"
        }

    position_data = {

        "direction": direction,
        "quantity": actual_size,
        "entry": entry_price,
        "risk_R": r_value,
        "stop": final_stop,

        "tp_order_id": tp_order_id,

    }

class SignalHandler(BaseHTTPRequestHandler):

    def log_message(self, format, *args):
        return

    def send_json(self, status_code, payload):
        try:
            body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
            self.send_response(status_code)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
        except Exception:
            pass

    def do_GET(self):
        if self.path == "/health":
            with state_lock:
                positions = get_all_active_positions()
                payload = {
                    "platform": "REZA EXECUTION PLATFORM",
                    "version": VERSION,
                    "bot_version": BOT_VERSION,
                    "symbol": DISPLAY_SYMBOL,
                    "status": "ONLINE",
                    "mode": "SCALP",
                    "stop_r": STOP_R_MULTIPLE,
                    "tp_r": TP_FULL_R,
                    "fixed_risk_usdt": FIXED_RISK_USDT,
                    "active_positions_count": len(positions),
                    "execution_status": state.get("execution_status")
                }
            self.send_json(200, payload)
            return
        self.send_json(404, {"error": "NOT_FOUND"})

    def do_POST(self):
        if self.path != BOT_SIGNAL_PATH:
            self.send_json(404, {"accepted": False, "error": "NOT_FOUND"})
            return

        try:
            length = int(self.headers.get("Content-Length", "0"))
            raw = self.rfile.read(length)
            signal = json.loads(raw.decode("utf-8"))
            validate_signal(signal)
            result = execute_signal(signal)
            self.send_json(200, result)
        except Exception as exc:
            self.send_json(400, {
                "accepted": False,
                "status": "REJECTED",
                "error": str(exc)
            })

class ReusableThreadingTCPServer(ThreadingTCPServer):
    allow_reuse_address = True
    daemon_threads = True


def start_signal_server():
    global server

    try:

            (PLATFORM_HOST, PLATFORM_PORT),

            print()

        target=server.serve_forever,
