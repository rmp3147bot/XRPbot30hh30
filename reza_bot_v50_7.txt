def utc_now():
    return datetime.now(timezone.utc)


def parse_xt_timestamp(value):
    if value is None:
        return None

    try:
        numeric_value = float(value)
    except (TypeError, ValueError):
        return None

    if not math.isfinite(numeric_value):
        return None

    if numeric_value > 100000000000:
        numeric_value /= 1000.0

    try:
        return datetime.fromtimestamp(numeric_value, timezone.utc)
    except (OverflowError, OSError, ValueError):
        return None


def parse_utc_datetime(value):
    if value is None:
        return None

    if isinstance(value, datetime):
        if value.tzinfo is None:
            return value.replace(tzinfo=timezone.utc)
        return value.astimezone(timezone.utc)

    if isinstance(value, (int, float)):
        return parse_xt_timestamp(value)

    if isinstance(value, str):
        text = value.strip()
        if not text:
            return None

        try:
            numeric_value = float(text)
            return parse_xt_timestamp(numeric_value)
        except (TypeError, ValueError):
            pass

        if text.endswith(" UTC"):
            text = text[:-4]

        try:
            parsed = datetime.strptime(text, "%Y-%m-%d %H:%M:%S")
            return parsed.replace(tzinfo=timezone.utc)
        except ValueError:
            pass

        try:
            parsed = datetime.fromisoformat(text)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=timezone.utc)
            return parsed.astimezone(timezone.utc)
        except ValueError:
            pass

    return None


def format_utc(dt):
    if dt is None:
        return ""
    parsed = parse_utc_datetime(dt)
    if parsed is None:
        return ""
    return parsed.strftime("%Y-%m-%d %H:%M:%S UTC")


def restore_signal_datetime_fields(signal):
    if not isinstance(signal, dict):
        return signal

    for field in ("time", "source_candle"):
        if field not in signal:
            continue
        value = signal.get(field)
        parsed = parse_utc_datetime(value)
        if parsed is not None:
            signal[field] = parsed

    return signal

def session_start_for(dt):
    dt = parse_utc_datetime(dt)
    if dt is None:
        return None

    base = dt.replace(minute=0, second=0, microsecond=0)
    
    if dt.minute >= SESSION_MINUTE:
        return base.replace(minute=SESSION_MINUTE)
    else:
        return (base - timedelta(hours=1)).replace(minute=SESSION_MINUTE)


def next_session_start_after(dt):
    current = session_start_for(dt)
    if current is None:
        return None
    if current > dt:
        return current
    return current + timedelta(hours=1)


def session_end_for(start):
    start = parse_utc_datetime(start)
    if start is None:
        return None
    return start + timedelta(hours=1)


def candle_session_start(candle_time):
    return session_start_for(candle_time)

def atomic_write_json(path, data):
    ensure_data_dir()
    temporary_file = path + ".tmp"

    try:
        with open(temporary_file, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2, default=str)
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temporary_file, path)

    except Exception as exc:
        log.error("Could not write JSON state %s: %s", path, exc)
        try:
            if os.path.exists(temporary_file):
                os.remove(temporary_file)
        except Exception:
            pass


def read_json_file(path, default):
    if not os.path.exists(path):
        return default

    try:
        with open(path, "r", encoding="utf-8") as f:
            data = json.load(f)
        return data

    except Exception as exc:
        log.warning("Could not read JSON state %s: %s", path, exc)
        return default

def restore_session_signal_lock():
    global session_state, touch_state

    data = read_json_file(SESSION_STATE_FILE, {})

    if not isinstance(data, dict):
        return

    saved_start = data.get("start")
    if not saved_start:
        return

    saved_dt = parse_utc_datetime(saved_start)
    if saved_dt is None:
        return

    now = utc_now()
    current_session_start = session_start_for(now)

    if current_session_start != saved_dt:
        log.info("🔄 NEW SESSION DETECTED - RESETTING HYPER AND SIGNAL LOCK")
        log.info(f"   OLD SESSION: {format_utc(saved_dt)}")
        log.info(f"   NEW SESSION: {format_utc(current_session_start)}")

        if session_state is not None:
            session_state["signal_generated"] = False
            session_state["signal_count"] = 0
            session_state["signal_id"] = None
            session_state["hyper_locked"] = False
            session_state["ready"] = False
            session_state["hyper"] = {
                "high": None,
                "low": None,
                "r": None,
                "source": "",
                "source_candle_time": None
            }
            session_state["start"] = current_session_start
            session_state["end"] = session_end_for(current_session_start)
            session_state["boundary_candle_time"] = None
            session_state["analysis_candles"] = 0
            
            if os.path.exists(SESSION_STATE_FILE):
                os.remove(SESSION_STATE_FILE)
            
            save_current_session_state()
            
            touch_state = {
                "pending_touch": None,
                "pending_touch_price": None,
                "pending_touch_time": None,
                "candle_checked": False,
                "last_checked_candle": None,
                "filters_passed": False
            }
        return

    if not data.get("signal_generated", False):
        return

    if session_state is None:
        return

    session_state["signal_generated"] = True
    session_state["signal_count"] = data.get("signal_count", 1)
    session_state["signal_id"] = data.get("signal_id")
    session_state["hyper_locked"] = bool(data.get("hyper_locked", False))

    hyper = session_state.setdefault("hyper", {})
    if data.get("hyper_high") is not None:
        hyper["high"] = data.get("hyper_high")
    if data.get("hyper_low") is not None:
        hyper["low"] = data.get("hyper_low")
    if data.get("r") is not None:
        hyper["r"] = float(data.get("r"))

    log.info("SESSION SIGNAL LOCK RESTORED: %s (count=%s)", data.get("signal_id"), session_state["signal_count"])

def read_platform_queue():
    ensure_platform_queue_file()

    signals = []

    try:
        with open(PLATFORM_QUEUE_FILE, "r", encoding="utf-8") as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue

                try:
                    signal = json.loads(line)
                    if not isinstance(signal, dict):
                        continue

                    signal = restore_signal_datetime_fields(signal)
                    signal_id = signal.get("signal_id")

                    if signal_id:
                        signals.append(signal)

                except Exception:
                    log.warning("Invalid platform queue line ignored.")

    except Exception as exc:
        log.error("Could not read platform queue: %s", exc)

    unique = []
    seen = set()

    for signal in signals:
        signal_id = str(signal.get("signal_id"))
        if signal_id in seen:
            continue
        seen.add(signal_id)
        unique.append(signal)

    return unique


def write_platform_queue(signals):
    ensure_platform_queue_file()

    temporary_file = PLATFORM_QUEUE_FILE + ".tmp"

    try:
        with open(temporary_file, "w", encoding="utf-8") as f:
            for signal in signals:
                f.write(json.dumps(signal, ensure_ascii=False, default=str) + "\n")
            f.flush()
            try:
                os.fsync(f.fileno())
            except Exception:
                pass
        os.replace(temporary_file, PLATFORM_QUEUE_FILE)

    except Exception as exc:
        log.error("Could not write platform queue: %s", exc)
        try:
            if os.path.exists(temporary_file):
                os.remove(temporary_file)
        except Exception:
            pass


def refresh_pending_count():
    signals = read_platform_queue()

    with platform_lock:
        platform_state["pending_count"] = len(signals)

    return len(signals)


def queue_platform_signal(signal):
    signals = read_platform_queue()

    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))

    existing_ids = {str(x.get("signal_id")) for x in signals}

    if signal_id not in existing_ids:
        signals.append(signal)

    write_platform_queue(signals)

    with platform_lock:
        platform_state["pending_count"] = len(signals)

def validate_platform_ack(data, sent_signal_id):
    if not isinstance(data, dict):
        return False

    returned_signal_id = data.get("signal_id")
    if returned_signal_id is None:
        return False

    if str(returned_signal_id) != str(sent_signal_id):
        log.error("INVALID PLATFORM ACK: signal_id mismatch. sent=%s returned=%s",
                  sent_signal_id, returned_signal_id)
        return False

    ack = bool(data.get("ack", False))
    status = str(data.get("status", "")).lower()
    already_received = bool(data.get("already_received", False))
    already_accepted = bool(data.get("already_accepted", False))

    if ack or already_received or already_accepted:
        return True

    if status in ("already_received", "already_accepted"):
        return True

    return False

def send_signal_to_platform(signal):
    if not PLATFORM_ENABLED:
        update_signal_status(
            signal,
            "WAITING_FOR_PLATFORM",
            "WAITING_FOR_PLATFORM",
            "PLATFORM_DISABLED"
        )
        return {"success": False, "status": "PLATFORM_DISABLED"}

    signal = restore_signal_datetime_fields(signal)
    signal_id = str(signal.get("signal_id"))
    payload = build_platform_payload(signal)

    with platform_delivery_lock:
        try:
            response = platform_http.post(
                PLATFORM_BASE_URL + PLATFORM_SIGNAL_PATH,
                json=payload,
                timeout=PLATFORM_TIMEOUT
            )

            if response.status_code not in (200, 201, 202):
                raise RuntimeError(f"HTTP {response.status_code}: {response.text[:1000]}")

            try:
                data = response.json()
            except Exception:
                data = {}

            if not validate_platform_ack(data, signal_id):
                raise RuntimeError(f"Platform response did not contain valid ACK: {data}")

            with platform_lock:
                platform_state["online"] = True
                platform_state["last_success"] = utc_now()
                platform_state["last_error"] = None
                platform_state["last_ack_signal"] = signal_id

            remove_platform_signal(signal_id)

            update_signal_status(
                signal,
                "READY_FOR_EXECUTION",
                "ACKNOWLEDGED",
                "PLATFORM_ACK",
                response=data
            )

            return {"success": True, "status": data.get("status", "ACKNOWLEDGED"), "response": data}

        except Exception as exc:
            with platform_lock:
                platform_state["online"] = False
                platform_state["last_error"] = str(exc)

            queue_platform_signal(signal)

            update_signal_status(
                signal,
                "WAITING_FOR_PLATFORM",
                "WAITING_FOR_PLATFORM",
                "DELIVERY_FAILED_PENDING",
                response={"error": str(exc)}
            )

            return {"success": False, "status": "WAITING_FOR_PLATFORM", "error": str(exc)}

    refresh_pending_count()

def floor_to_tick(value, tick):
    try:
        val = float(value)
        tick_val = float(tick)
        if tick_val <= 0:
            return val
        return math.floor(val / tick_val) * tick_val
    except Exception:
        return value

def get_ticker():
    url = f"{XT_BASE_URL}/future/market/v1/public/q/agg-ticker"
    params = {"symbol": SYMBOL}

    try:
        response = requests.get(url, params=params, timeout=10)
        response.raise_for_status()

    except Exception as exc:
        raise RuntimeError(f"XT ticker request failed for {SYMBOL}: {exc}")

    try:
        data = response.json()

    except Exception as exc:
        raise RuntimeError(f"XT returned non-JSON ticker for {SYMBOL}: {response.text[:1000]}") from exc

    if not isinstance(data, dict):
        raise RuntimeError(f"XT returned invalid ticker for {SYMBOL}: {repr(data)[:1500]}")

    result = data.get("result")

    if isinstance(result, list):
        if not result:
            raise RuntimeError(f"XT returned empty ticker for {SYMBOL}: {repr(data)[:1500]}")
        result = result[0]

    elif not isinstance(result, dict):
        raise RuntimeError(f"XT returned invalid ticker result for {SYMBOL}: {repr(data)[:1500]}")

    last = result.get("c") if result.get("c") is not None else result.get("last")
    if last is None:
        last = result.get("lastPrice") if result.get("lastPrice") is not None else result.get("price")

    if last is None:
        raise RuntimeError(f"XT ticker has no last price for {SYMBOL}: {repr(data)[:1500]}")

    try:
        last = float(last)

    except (TypeError, ValueError) as exc:
        raise RuntimeError(f"XT ticker returned invalid last price for {SYMBOL}: {repr(last)}") from exc

    if not math.isfinite(last) or last <= 0:
        raise RuntimeError(f"XT ticker returned invalid last price for {SYMBOL}: {repr(last)}")

    ts = result.get("ts") if result.get("ts") is not None else result.get("timestamp")
    if ts is None:
        ts = result.get("time") if result.get("time") is not None else result.get("t")

    return {"last": last, "ts": ts, "raw": data}

# ATR CALCULATION
# ============================================================================

def calculate_atr(candles, period=5):
    if len(candles) < 2:
        return None
    
    available_period = min(period, len(candles) - 1)
    
    if available_period < 1:
        return None
    
    tr_list = []
    for i in range(1, len(candles)):
        high = candles[i]['high']
        low = candles[i]['low']
        prev_close = candles[i-1]['close']
        
        tr = max(high - low, abs(high - prev_close), abs(low - prev_close))
        tr_list.append(tr)
    
    if not tr_list:
        return None
    
    atr = sum(tr_list[-available_period:]) / available_period
    
    log.info(f"📊 ATR CALCULATED: period={available_period}/{period}, atr={atr:.4f}")
    
    return atr


# ============================================================================
# ✅ DYNAMIC R (محاسبه R از برگشت قیمت)
# ============================================================================

def get_dynamic_r(candles, lookback=5):
    if len(candles) < lookback:
        return None
    
    try:
        recent = candles[-lookback:]
        highs = [float(c['high']) for c in recent]
        lows = [float(c['low']) for c in recent]
    except (ValueError, TypeError, KeyError):
        return None
    
    recent_high = max(highs)
    recent_low = min(lows)
    dynamic_r = recent_high - recent_low
    
    atr = calculate_atr(candles, ATR_PERIOD)
    if atr is None or atr <= 0:
        return dynamic_r
    
    if dynamic_r < atr * 0.3:
        r_used = atr * 0.5
        log.info(f"📊 DYNAMIC R: range={dynamic_r:.4f} too small, using ATR*0.5={r_used:.4f}")
        return r_used
    elif dynamic_r < atr * 0.5:
        r_used = atr * 0.7
        log.info(f"📊 DYNAMIC R: range={dynamic_r:.4f} small, using ATR*0.7={r_used:.4f}")
        return r_used
    
    log.info(f"📊 DYNAMIC R: using range={dynamic_r:.4f} from last {lookback} candles")
    return dynamic_r

def is_range_market(candles, lookback=10):
    if len(candles) < lookback:
        return False
    
    try:
        highs = [float(c['high']) for c in candles[-lookback:]]
        lows = [float(c['low']) for c in candles[-lookback:]]
    except (ValueError, TypeError, KeyError):
        return False
    
    range_high = max(highs)
    range_low = min(lows)
    range_size = range_high - range_low
    
    atr = calculate_atr(candles, ATR_PERIOD)
    if atr is None or atr <= 0:
        return False
    
    if range_size < atr * RANGE_THRESHOLD:
        log.info(f"📊 RANGE DETECTED: range={range_size:.4f}, atr*{RANGE_THRESHOLD}={atr*RANGE_THRESHOLD:.4f}")
        return True
    
    return False


# ============================================================================
# ✅ RSI CALCULATION
# ============================================================================

def calculate_rsi(candles, period=6):
    if len(candles) < period + 1:
        return None

    recent_candles = candles[-(period + 1):]
    
    gains = []
    losses = []
    
    for i in range(1, len(recent_candles)):
        change = recent_candles[i]['close'] - recent_candles[i-1]['close']
        if change >= 0:
            gains.append(change)
            losses.append(0)
        else:
            gains.append(0)
            losses.append(-change)
    
    if not gains or not losses:
        return None
    
    avg_gain = sum(gains) / period
    avg_loss = sum(losses) / period
    
    if avg_loss == 0:
        return 100.0
    
    rs = avg_gain / avg_loss
    rsi = 100 - (100 / (1 + rs))
    
    return rsi

def setup_session_locked_levels(state, candle):
    if state is None or candle is None:
        return False

    if candle.get("confirmed", 0) != 1:
        log.debug("Candle not confirmed, skipping")
        return False

    candle_time = candle.get("time")
    if candle_time is None:
        log.error("SESSION INIT FAILED: candle time missing.")
        return False

    expected_session_start = session_start_for(candle_time)
    if expected_session_start != state.get("start"):
        log.debug(f"Candle {format_utc(candle_time)} not in current session")
        return False

    if state.get("hyper_locked", False):
        log.warning("⚠️ HYPER ALREADY LOCKED, skipping setup")
        return True

    hyper_high = float(candle["high"])
    hyper_low = float(candle["low"])
    r = hyper_high - hyper_low

    if hyper_high < hyper_low:
        log.error("SESSION INIT FAILED: invalid Hyper range | high=%s low=%s", hyper_high, hyper_low)
        return False

    if r < MIN_R_VALUE:
        log.info(f"❌ R FILTER FAILED: r={r:.4f} < {MIN_R_VALUE} on candle {format_utc(candle_time)}")
        return False

    body = abs(candle["close"] - candle["open"])
    total_range = hyper_high - hyper_low
    if total_range > 0:
        body_percent = (body / total_range) * 100
    else:
        body_percent = 0

    if body_percent < MIN_BODY_PERCENT:
        log.info(f"❌ BODY FILTER FAILED: body_percent={body_percent:.2f}% < {MIN_BODY_PERCENT}% on candle {format_utc(candle_time)}")
        return False

    log.info(f"✅ ALL FILTERS PASSED: r={r:.4f}, body={body_percent:.2f}% on candle {format_utc(candle_time)}")

    state["ready"] = True
    state["analysis_candles"] = state.get("analysis_candles", 0) + 1
    state["structure"] = {}
    state["hyper"] = {
        "high": hyper_high,
        "low": hyper_low,
        "r": r,
        "source": f"SESSION_CANDLE_LOCKED_{format_utc(candle_time)}",
        "source_candle_time": candle_time
    }
    state["hyper_locked"] = True
    state["boundary_candle_time"] = candle_time

    save_session(state)
    save_current_session_state()

    log.info("✅ SESSION READY | start=%s | locked_from=%s | hyper_high=%s | hyper_low=%s | R=%s | body=%s%%",
             format_utc(state["start"]), format_utc(candle_time), hyper_high, hyper_low, r, body_percent)

    return True

# ============================================================================
# ✅ CHECK FILTERS ON CANDLE (با R پویا + فیلتر رنج)
# ============================================================================

def check_filters_on_candle(candle, direction, session_state):
    if candle is None:
        return False, None

    if session_state is None or not session_state.get("ready", False):
        return False, None

    # ================================================================
    # ✅ RANGE FILTER - جلوگیری از سیگنال در بازار رنج
    # ================================================================
    if USE_RANGE_FILTER and is_range_market(primary_candles):
        log.info(f"⏸️ RANGE MARKET DETECTED - SKIPPING {direction} SIGNAL")
        return False, None

    body = abs(candle["close"] - candle["open"])
    total_range = candle["high"] - candle["low"]
    
    if total_range == 0 or body / total_range < 0.1:
        return False, None
    
    upper_shadow = candle["high"] - max(candle["close"], candle["open"])
    lower_shadow = min(candle["close"], candle["open"]) - candle["low"]
    is_bullish = candle["close"] > candle["open"]

    is_hammer = is_bullish and lower_shadow >= SHADOW_MULTIPLIER * body and upper_shadow <= body
    is_shooting_star = not is_bullish and upper_shadow >= SHADOW_MULTIPLIER * body and lower_shadow <= body
    
    is_strong_hammer = is_bullish and lower_shadow >= STRONG_PATTERN_MULTIPLIER * body and upper_shadow <= body
    is_strong_shooting_star = not is_bullish and upper_shadow >= STRONG_PATTERN_MULTIPLIER * body and lower_shadow <= body

    rsi = calculate_rsi(primary_candles, 6)
    
    if direction == "LONG":
        if not (is_hammer or is_strong_hammer):
            log.info(f"   ❌ FILTER: Not a Hammer pattern")
            return False, None
        if rsi is not None and rsi >= RSI_OVERSOLD:
            log.info(f"   ❌ FILTER: RSI={rsi:.1f} >= {RSI_OVERSOLD}")
            return False, None
        log.info(f"   ✅ FILTER: Hammer pattern detected, RSI={rsi:.1f}")
        
    elif direction == "SHORT":
        if not (is_shooting_star or is_strong_shooting_star):
            log.info(f"   ❌ FILTER: Not a Shooting Star pattern")
            return False, None
        if rsi is not None and rsi <= RSI_OVERBOUGHT:
            log.info(f"   ❌ FILTER: RSI={rsi:.1f} <= {RSI_OVERBOUGHT}")
            return False, None
        log.info(f"   ✅ FILTER: Shooting Star pattern detected, RSI={rsi:.1f}")

    volume_ratio = None
    is_strong_pattern = False
    
    if len(primary_candles) >= 6:
        last_5_volumes = [c.get("samples", 0) for c in primary_candles[-6:-1]]
        avg_volume = sum(last_5_volumes) / len(last_5_volumes) if last_5_volumes else 1
        current_volume = candle.get("samples", 0)
        volume_ratio = current_volume / avg_volume if avg_volume > 0 else 0
        
        if (direction == "LONG" and is_strong_hammer) or (direction == "SHORT" and is_strong_shooting_star):
            is_strong_pattern = True
            log.info(f"   ✅ FILTER: STRONG PATTERN - volume ignored")
        else:
            if volume_ratio < VOLUME_SPIKE_THRESHOLD:
                log.info(f"   ❌ FILTER: Volume too low ({volume_ratio:.2f}x < {VOLUME_SPIKE_THRESHOLD}x)")
                return False, None
            log.info(f"   ✅ FILTER: Volume OK ({volume_ratio:.2f}x)")

    # ================================================================
    # ✅ DYNAMIC R - محاسبه R از برگشت قیمت
    # ================================================================
    if USE_DYNAMIC_R:
        dynamic_r = get_dynamic_r(primary_candles, DYNAMIC_R_LOOKBACK)
        if dynamic_r is not None and dynamic_r > 0:
            if 'dynamic_r' not in session_state:
                session_state['dynamic_r'] = {}
            session_state['dynamic_r']['value'] = dynamic_r
            session_state['dynamic_r']['updated_at'] = utc_now()
            log.info(f"   ✅ DYNAMIC R: {dynamic_r:.4f} from last {DYNAMIC_R_LOOKBACK} candles")
        else:
            hyper = session_state["hyper"]
            dynamic_r = hyper["r"]
            log.info(f"   ⚠️ DYNAMIC R failed, using hyper R: {dynamic_r:.4f}")
        
        if dynamic_r < MIN_R_VALUE:
            log.info(f"   ❌ FILTER: R={dynamic_r:.4f} < {MIN_R_VALUE}")
            return False, None
    else:
        hyper = session_state["hyper"]
        dynamic_r = hyper["r"]
        if dynamic_r < MIN_R_VALUE:
            log.info(f"   ❌ FILTER: R={dynamic_r:.4f} < {MIN_R_VALUE}")
            return False, None

    body_percent = (body / total_range) * 100 if total_range > 0 else 0
    if body_percent < MIN_BODY_PERCENT:
        log.info(f"   ❌ FILTER: Body={body_percent:.1f}% < {MIN_BODY_PERCENT}%")
        return False, None

    log.info(f"   ✅ ALL FILTERS PASSED on candle {format_utc(candle['time'])}")
    return True, {
        "volume_ratio": volume_ratio,
        "rsi": rsi,
        "body_percent": body_percent,
        "is_strong_pattern": is_strong_pattern,
        "candle_time": candle["time"],
        "r_used": dynamic_r
    }

def detect_trend_signals(candle, session_state):
    """
    تشخیص سیگنال‌های روندی (Trend Following)
    - BREAKOUT LONG: Close > Hyper_High * 1.001 + Body ≥ 40%
    - BREAKDOWN SHORT: Close < Hyper_Low * 0.999 + Body ≥ 40%
    - RSI بین ۴۵-۷۵ برای LONG، ۲۵-۵۵ برای SHORT
    - حجم ≥ ۰.۷×میانگین
    """
    if candle is None or session_state is None:
        return None
    
    if not session_state.get("ready", False):
        return None
    
    if session_state.get("signal_count", 0) >= MAX_SIGNALS_PER_SESSION:
        return None
    
    # ================================================================
    # ✅ RANGE FILTER - در بازار رنج سیگنال روندی نمی‌دهیم
    # ================================================================
    if USE_RANGE_FILTER and is_range_market(primary_candles):
        return None
    
    hyper = session_state.get("hyper", {})
    hyper_high = hyper.get("high")
    hyper_low = hyper.get("low")
    
    if hyper_high is None or hyper_low is None:
        return None
    
    r = hyper_high - hyper_low
    if r <= 0:
        return None
    
    body = abs(candle["close"] - candle["open"])
    total_range = candle["high"] - candle["low"]
    
    if total_range == 0:
        return None
    
    body_percent = (body / total_range) * 100
    is_bullish = candle["close"] > candle["open"]
    rsi = calculate_rsi(primary_candles, 6)
    
    volume_ratio = None
    if len(primary_candles) >= 6:
        last_5_volumes = [c.get("samples", 0) for c in primary_candles[-6:-1]]
        avg_volume = sum(last_5_volumes) / len(last_5_volumes) if last_5_volumes else 1
        current_volume = candle.get("samples", 0)
        volume_ratio = current_volume / avg_volume if avg_volume > 0 else 0
    else:
        volume_ratio = 1.0

    # ================================================================
    # سیگنال LONG روندی (Breakout)
    # ================================================================
    if is_bullish and candle["close"] > hyper_high * (1 + TREND_BREAKOUT_THRESHOLD):
        if body_percent < TREND_MIN_BODY_PERCENT:
            log.info(f"📈 TREND LONG FAILED: body={body_percent:.1f}% < {TREND_MIN_BODY_PERCENT}%")
            return None
        
        if rsi is not None and (rsi < TREND_MIN_RSI or rsi > TREND_MAX_RSI):
            log.info(f"📈 TREND LONG FAILED: RSI={rsi:.1f} (range: {TREND_MIN_RSI}-{TREND_MAX_RSI})")
            return None
        
        if volume_ratio is not None and volume_ratio < TREND_MIN_VOLUME_RATIO:
            log.info(f"📈 TREND LONG FAILED: volume={volume_ratio:.2f}x < {TREND_MIN_VOLUME_RATIO}x")
            return None
        
        log.info(f"📈 TREND BREAKOUT LONG: close={candle['close']:.4f}, body={body_percent:.1f}%, RSI={rsi:.1f}, vol={volume_ratio:.2f}x")
        
        if USE_DYNAMIC_R:
            dynamic_r = get_dynamic_r(primary_candles, DYNAMIC_R_LOOKBACK)
            if dynamic_r is not None and dynamic_r > 0:
                r = max(r, dynamic_r)
        
        return {
            "direction": "LONG",
            "entry": candle["close"],
            "candle": candle,
            "hyper_high": hyper_high,
            "hyper_low": hyper_low,
            "r": r,
            "touch_type": "TREND_BREAKOUT_LONG",
            "volume_ratio": volume_ratio,
            "rsi": rsi,
            "body_percent": body_percent,
            "is_strong_pattern": True
        }
    
    # ================================================================
    # سیگنال SHORT روندی (Breakdown)
    # ================================================================
    if not is_bullish and candle["close"] < hyper_low * (1 - TREND_BREAKOUT_THRESHOLD):
        if body_percent < TREND_MIN_BODY_PERCENT:
            log.info(f"📉 TREND SHORT FAILED: body={body_percent:.1f}% < {TREND_MIN_BODY_PERCENT}%")
            return None
        
        if rsi is not None and (rsi < TREND_SHORT_MIN_RSI or rsi > TREND_SHORT_MAX_RSI):
            log.info(f"📉 TREND SHORT FAILED: RSI={rsi:.1f} (range: {TREND_SHORT_MIN_RSI}-{TREND_SHORT_MAX_RSI})")
            return None
        
        if volume_ratio is not None and volume_ratio < TREND_MIN_VOLUME_RATIO:
            log.info(f"📉 TREND SHORT FAILED: volume={volume_ratio:.2f}x < {TREND_MIN_VOLUME_RATIO}x")
            return None
        
        log.info(f"📉 TREND BREAKDOWN SHORT: close={candle['close']:.4f}, body={body_percent:.1f}%, RSI={rsi:.1f}, vol={volume_ratio:.2f}x")
        
        if USE_DYNAMIC_R:
            dynamic_r = get_dynamic_r(primary_candles, DYNAMIC_R_LOOKBACK)
            if dynamic_r is not None and dynamic_r > 0:
                r = max(r, dynamic_r)
        
        return {
            "direction": "SHORT",
            "entry": candle["close"],
            "candle": candle,
            "hyper_high": hyper_high,
            "hyper_low": hyper_low,
            "r": r,
            "touch_type": "TREND_BREAKDOWN_SHORT",
            "volume_ratio": volume_ratio,
            "rsi": rsi,
            "body_percent": body_percent,
            "is_strong_pattern": True
        }
    
    return None

def detect_signal(candle, session_state):
    """
    تشخیص سیگنال هیبرید (روند + برگشت)
    """
    if candle is None:
        return None

    if candle.get("confirmed", 0) != 1:
        return None

    if session_state is None or not session_state.get("ready", False):
        return None

    if session_state.get("signal_count", 0) >= MAX_SIGNALS_PER_SESSION:
        log.info(f"⏸️ MAX SIGNALS REACHED: {session_state['signal_count']}/{MAX_SIGNALS_PER_SESSION}")
        return None

    # ================================================================
    # ✅ RANGE FILTER - جلوگیری از سیگنال در بازار رنج
    # ================================================================
    if USE_RANGE_FILTER and is_range_market(primary_candles):
        log.info(f"⏸️ RANGE MARKET DETECTED - SKIPPING ALL SIGNALS")
        return None

    # ================================================================
    # ۱. سیگنال برگشتی (Reversal)
    # ================================================================
    reversal = detect_reversal_signals(candle, session_state)
    if reversal:
        log.info(f"🔄 REVERSAL SIGNAL: {reversal['direction']}")
        return reversal
    
    # ================================================================
    # ۲. سیگنال روندی (Trend)
    # ================================================================
    trend = detect_trend_signals(candle, session_state)
    if trend:
        log.info(f"📈 TREND SIGNAL: {trend['direction']}")
        return trend
    
    return None


def detect_reversal_signals(candle, session_state):
    """
    تشخیص سیگنال‌های برگشتی (Hammer / Shooting Star)
    """
    body = abs(candle["close"] - candle["open"])
    total_range = candle["high"] - candle["low"]
    
    if total_range == 0 or body / total_range < 0.1:
        return None
    
    upper_shadow = candle["high"] - max(candle["close"], candle["open"])
    lower_shadow = min(candle["close"], candle["open"]) - candle["low"]
    is_bullish = candle["close"] > candle["open"]

    is_hammer = is_bullish and lower_shadow >= SHADOW_MULTIPLIER * body and upper_shadow <= body
    is_shooting_star = not is_bullish and upper_shadow >= SHADOW_MULTIPLIER * body and lower_shadow <= body
    
    if not is_hammer and not is_shooting_star:
        return None
    
    is_strong_hammer = is_bullish and lower_shadow >= STRONG_PATTERN_MULTIPLIER * body and upper_shadow <= body
    is_strong_shooting_star = not is_bullish and upper_shadow >= STRONG_PATTERN_MULTIPLIER * body and lower_shadow <= body

    rsi = calculate_rsi(primary_candles, 6)
    
    if is_hammer and (rsi is None or rsi >= RSI_OVERSOLD):
        return None
    
    if is_shooting_star and (rsi is None or rsi <= RSI_OVERBOUGHT):
        return None

    volume_ratio = None
    is_strong_pattern = False
    
    if len(primary_candles) >= 6:
        last_5_volumes = [c.get("samples", 0) for c in primary_candles[-6:-1]]
        avg_volume = sum(last_5_volumes) / len(last_5_volumes) if last_5_volumes else 1
        current_volume = candle.get("samples", 0)
        volume_ratio = current_volume / avg_volume if avg_volume > 0 else 0
        
        if (is_hammer and is_strong_hammer) or (is_shooting_star and is_strong_shooting_star):
            is_strong_pattern = True
            log.info(f"🔵 STRONG PATTERN DETECTED: volume_ratio={volume_ratio:.2f}x (ignored)")
        else:
            if volume_ratio < VOLUME_SPIKE_THRESHOLD:
                return None

    hyper = session_state["hyper"]
    r = hyper["r"]
    
    if r < MIN_R_VALUE:
        return None
    
    body_percent = (body / total_range) * 100 if total_range > 0 else 0
    if body_percent < MIN_BODY_PERCENT:
        return None

    if is_hammer:
        return {
            "direction": "LONG",
            "entry": candle["close"],
            "candle": candle,
            "hyper_high": hyper["high"],
            "hyper_low": hyper["low"],
            "r": r,
            "touch_type": "MOMENTUM_REVERSAL_HAMMER" + ("_STRONG" if is_strong_pattern else ""),
            "volume_ratio": volume_ratio,
            "rsi": rsi,
            "body_percent": body_percent,
            "is_strong_pattern": is_strong_pattern
        }
    
    if is_shooting_star:
        return {
            "direction": "SHORT",
            "entry": candle["close"],
            "candle": candle,
            "hyper_high": hyper["high"],
            "hyper_low": hyper["low"],
            "r": r,
            "touch_type": "MOMENTUM_REVERSAL_SHOOTING_STAR" + ("_STRONG" if is_strong_pattern else ""),
            "volume_ratio": volume_ratio,
            "rsi": rsi,
            "body_percent": body_percent,
            "is_strong_pattern": is_strong_pattern
        }

    return None

# BUILD REFERENCE SIGNAL (با R پویا و TP=1R)
# ============================================================================

def build_reference_signal(strategy_signal, breakout_candle):
    if strategy_signal is None or breakout_candle is None:
        return None

    direction = strategy_signal.get("direction")
    if direction not in ("LONG", "SHORT"):
        return None

    hyper_high = float(strategy_signal["hyper_high"])
    hyper_low = float(strategy_signal["hyper_low"])
    hyper_r = float(strategy_signal.get("r", hyper_high - hyper_low))
    
    if hyper_r <= 0:
        hyper_r = float(hyper_high - hyper_low)
        if hyper_r <= 0:
            return None

    entry = float(strategy_signal["entry"])

    # ================================================================
    # ✅ دریافت R پویا از session_state
    # ================================================================
    r_used = None
    if USE_DYNAMIC_R and session_state and 'dynamic_r' in session_state:
        r_used = session_state['dynamic_r'].get('value')
        if r_used is not None and r_used > 0:
            log.info(f"📊 USING DYNAMIC R: {r_used:.4f}")
    
    if r_used is None or r_used <= 0:
        r_used = hyper_r
        log.warning(f"⚠️ Using hyper R: {r_used:.4f}")

    breakout_time = breakout_candle["time"]
    signal_id = f"{BOT_ID}-{direction}-{breakout_time.strftime('%Y%m%d%H%M')}"

    stop_distance = r_used
    
    atr = calculate_atr(primary_candles, ATR_PERIOD)
    if USE_ATR_STOP and atr is not None and atr > 0:
        stop_distance = atr * ATR_MULTIPLIER
        log.info(f"📊 STOP CALCULATION: atr={atr:.4f}, multiplier={ATR_MULTIPLIER}, stop_distance={stop_distance:.4f}")
    else:
        log.warning("⚠️ ATR unavailable, using R for stop")
    
    min_stop = r_used * 0.5
    max_stop = r_used * 2.0
    
    if stop_distance < min_stop:
        stop_distance = min_stop
        log.warning(f"⚠️ STOP DISTANCE TOO SMALL, adjusted to {min_stop:.4f}")
    elif stop_distance > max_stop:
        stop_distance = max_stop
        log.warning(f"⚠️ STOP DISTANCE TOO LARGE, adjusted to {max_stop:.4f}")
    
    if direction == "LONG":
        initial_stop = entry - stop_distance
        initial_tp = entry + (r_used * 1.0)  # ✅ TP = 1R
    else:
        initial_stop = entry + stop_distance
        initial_tp = entry - (r_used * 1.0)  # ✅ TP = 1R

    signal = {
        "signal_id": signal_id,
        "time": utc_now(),
        "source_candle": breakout_time,
        "symbol": SYMBOL,
        "reference_market": "XT XRPUSDT FUTURES",
        "direction": direction,
        "reference_entry": float(entry),
        "hyper_high": float(hyper_high),
        "hyper_low": float(hyper_low),
        "r": float(r_used),
        "initial_risk": float(r_used),
        "initial_stop": float(initial_stop),
        "initial_tp": float(initial_tp),
        "tp_r_multiple": 1.0,  # ✅ TP = 1R
        "entry_type": strategy_signal.get("touch_type", "MOMENTUM_REVERSAL"),
        "touch_price": float(strategy_signal["entry"]),
        "touch_type": strategy_signal.get("touch_type", ""),
        "bot_id": BOT_ID,
        "atr_used": float(atr) if atr is not None else None,
        "stop_distance": float(stop_distance),
        "candles_used": len(primary_candles),
        "volume_ratio": strategy_signal.get("volume_ratio"),
        "rsi": strategy_signal.get("rsi"),
        "body_percent": strategy_signal.get("body_percent"),
        "is_strong_pattern": strategy_signal.get("is_strong_pattern", False),
        "hyper_r": float(hyper_r)
    }

    log.info("SIGNAL CREATED | id=%s | direction=%s | entry=%s | r=%s (dynamic) | stop_dist=%s | atr=%s",
             signal_id, direction, signal["reference_entry"], 
             signal["r"], signal["stop_distance"], signal["atr_used"])

    return signal

# ============================================================================

def process_live_market(market):
    global current_candle, last_bucket, last_price, market_state, primary_candles
    global session_state, last_session_start, startup_waiting_for_session, touch_state

    price = float(market["last"])
    now = parse_utc_datetime(market.get("ts"))

    if now is None:
        now = utc_now()

    last_price = price
    market_state = market
    market_state["ts"] = now

    save_price(now, market["last"], market.get("mark"), market.get("index"), market.get("funding"))

    bucket = bucket_5m(now)

    if current_candle is None:
        current_candle = start_new_candle(bucket, price)
        last_bucket = bucket
        return

    if bucket == last_bucket:
        update_candle(current_candle, price)

    else:
        previous = close_candle(current_candle)
        previous["confirmed"] = 1

        if not primary_candles or previous["time"] > primary_candles[-1]["time"]:
            primary_candles.append(previous)

            if len(primary_candles) > MAX_PRIMARY_CANDLES:
                primary_candles = primary_candles[-MAX_PRIMARY_CANDLES:]

            save_closed_candle(previous)

class WebhookHandler(BaseHTTPRequestHandler):

    def log_message(self, format, *args):
        return

    def send_json(self, status_code, payload):
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self.send_response(status_code)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        if self.path == WEBHOOK_PATH:
            try:
                length = int(self.headers.get("Content-Length", "0"))
                raw = self.rfile.read(length)
                data = json.loads(raw.decode("utf-8"))

                event = data.get("event")
                signal_id = data.get("signal_id")

                log.info("📡 WEBHOOK RECEIVED: event=%s signal_id=%s", event, signal_id)

                append_webhook_log(data)

                print()
                print("=" * 78)
                print("📡 WEBHOOK RECEIVED FROM EXECUTION PLATFORM")
                print("=" * 78)
                print(f"EVENT           : {event}")
                print(f"SIGNAL ID       : {signal_id}")
                print(f"DIRECTION       : {data.get('direction')}")
                print(f"ENTRY PRICE     : {data.get('entry_price')}")
                print(f"EXIT PRICE      : {data.get('exit_price')}")
                print(f"P&L             : {data.get('pnl')} USDT")
                print(f"P&L %           : {data.get('pnl_percent')}%")
                print(f"REASON          : {data.get('reason')}")
                print(f"FINAL STEP      : {data.get('final_step')} ({(data.get('total_r', 0)):.1f}R)")
                print(f"CLOSE TIME      : {data.get('close_time')}")
                print("=" * 78)
                print()

                self.send_json(200, {
                    "status": "OK",
                    "message": "Webhook received",
                    "signal_id": signal_id
                })

            except Exception as exc:
                log.error("WEBHOOK HANDLER ERROR: %s", exc)
                self.send_json(400, {"status": "ERROR", "error": str(exc)})

        else:
            self.send_json(404, {"error": "NOT_FOUND"})

    def do_GET(self):
        if self.path == "/health":
            self.send_json(200, {
                "bot": f"REZA BOT {BOT_ID}",
                "version": VERSION,
                "project": PROJECT,
                "symbol": DISPLAY_SYMBOL,
                "status": "ONLINE",
                "signal_type": "HYBRID (TREND + REVERSAL)",
                "hyper_lock_mode": "FIRST_VALID_CANDLE",
                "touch_reset_on_return": True,
                "range_filter": USE_RANGE_FILTER,
                "range_threshold": RANGE_THRESHOLD,
                "dynamic_r": USE_DYNAMIC_R,
                "dynamic_r_lookback": DYNAMIC_R_LOOKBACK,
                "r_calculation": "DYNAMIC_R (from recent price action)",
                "max_signals_per_session": MAX_SIGNALS_PER_SESSION,
                "tp_r_multiple": 1.0,
                "volume_spike_threshold": VOLUME_SPIKE_THRESHOLD,
                "shadow_multiplier": SHADOW_MULTIPLIER,
                "strong_pattern_multiplier": STRONG_PATTERN_MULTIPLIER,
                "rsi_period": 6,
                "rsi_oversold": RSI_OVERSOLD,
                "rsi_overbought": RSI_OVERBOUGHT,
                "atr_multiplier": ATR_MULTIPLIER
            })
        else:
            self.send_json(404, {"error": "NOT_FOUND"})


def append_webhook_log(data):
    with open(WEBHOOK_LOG_FILE, "a", encoding="utf-8") as f:
        f.write(json.dumps(data, ensure_ascii=False, default=str) + "\n")


def start_webhook_server():
    try:
        server = ThreadingTCPServer((WEBHOOK_HOST, WEBHOOK_PORT), WebhookHandler)
        server.allow_reuse_address = True
        server.daemon_threads = True

        thread = threading.Thread(target=server.serve_forever, daemon=True, name="WEBHOOK-SERVER")
        thread.start()

        print()
        print("=" * 78)
        print("🌐 WEBHOOK SERVER STARTED")
        print("=" * 78)
        print(f"WEBHOOK LISTENER : http://{WEBHOOK_HOST}:{WEBHOOK_PORT}{WEBHOOK_PATH}")
        print(f"HEALTH           : http://{WEBHOOK_HOST}:{WEBHOOK_PORT}/health")
        print("=" * 78)

        return True

    except OSError as exc:
        if getattr(exc, "errno", None) == 98:
            print()
            print("=" * 78)
            print("⚠️ WEBHOOK SERVER: PORT 8765 IS ALREADY IN USE")
            print("=" * 78)
            print("Existing webhook server may be running.")
            print("=" * 78)
            return False
        raise
